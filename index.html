function createGRIInterestForm() { // 1. Create the form var form = FormApp.create('GRI Interest Form'); // Set description form.setDescription('As GRI continues to grow, we are looking to cultivate meaningful relationships with individuals who resonate with our mission. This form helps us learn more about your interests, skills, professional background, and networks so we can explore possible ways of collaborating and engaging together.'); // 2. Automatically collect email addresses form.setCollectEmail(true); // --- Form Questions --- // Question 1: Full Name form.addTextItem() .setTitle('Full name') .setRequired(true); // Question 3: Email Address (Backup/Verification field since collection is automated) form.addTextItem() .setTitle('Email address') .setRequired(true); // Question 4: Phone Number (with number validation) var phoneItem = form.addTextItem(); phoneItem.setTitle('Phone number'); var numberValidation = FormApp.createTextValidation() .requireNumber() .setHelpText('Please enter a valid phone number.') .build(); phoneItem.setValidation(numberValidation); // Question 5: Community / city / region form.addTextItem() .setTitle('Community / city / region') .setRequired(true); // Question 6: Why are you interested... form.addParagraphTextItem() .setTitle('Why are you interested in volunteering with us?') .setRequired(true); // Question 7: Skills Grid (Using a Grid Item for Yes/No/Explain) var skillsGrid = form.addGridItem(); skillsGrid.setTitle('Do you have skills in:') .setRows([ 'Social media skills (running campaigns, ads, etc.)', 'Marketing & communications', 'Fundraising', 'Developing trainings', 'Operations & Systems' ]) .setColumns(['Yes', 'No', 'Explain/Other details']) .setRequired(true); // Question 7 (Other Option) form.addTextItem() .setTitle('Other skills:'); // Question 8: Attach CV (Using the Paragraph Text workaround) form.addParagraphTextItem() .setTitle('Attach your CV (a must) - (Manually change this question type to File Upload in the Form Editor)') .setHelpText('Apps Script does not natively support creating File Upload questions. Please convert this to a File Upload question in your browser.') .setRequired(true); // --- Output URLs --- Logger.log('=================================================='); Logger.log('Form Created Successfully!'); Logger.log('Editor URL (To edit the form): ' + form.getEditUrl()); Logger.log('Published URL (To send to users): ' + form.getPublishedUrl()); Logger.log('=================================================='); }
